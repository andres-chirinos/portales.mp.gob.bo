toText = (v) => (v ?? "").toString().trim();

toInt = (v) => {
  const n = Number.parseInt(toText(v), 10);
  return Number.isFinite(n) ? n : null;
};

parseDenunciaFecha = (value) => {
  const text = toText(value);
  if (!text) return null;
  const [datePart, timePart = "00:00:00"] = text.split(" ");
  const [dd, mm, yyyy] = datePart.split("-").map(Number);
  const [hh = 0, min = 0, ss = 0] = timePart.split(":").map(Number);
  if (![dd, mm, yyyy].every(Number.isFinite)) return null;
  return new Date(yyyy, mm - 1, dd, hh, min, ss);
};

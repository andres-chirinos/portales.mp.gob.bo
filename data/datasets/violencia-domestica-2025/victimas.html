Victimas: perfil y condiciones
d3 = require("d3@7");
Plot = require("@observablehq/plot");

raw = await FileAttachment("VIOLENCIA_DOMESTICA 2025_publico__1.csv").csv({typed: false});

toText = (v) => (v ?? "").toString().trim();

toInt = (v) => {
  const n = Number.parseInt(toText(v), 10);
  return Number.isFinite(n) ? n : null;
};

rows = raw.map((d) => ({
  ...d,
  victima_sexo: toText(d.victima_sexo).toUpperCase(),
  victima_edad: toInt(d.victima_edad),
  victima_discapacidad: toText(d.victima_discapacidad),
  victima_nivel_educacion: toText(d.victima_nivel_educacion),
  hecho_departamento: toText(d.hecho_departamento)
}));

departamentos = Array.from(new Set(rows.map((d) => d.hecho_departamento).filter(Boolean))).sort((a, b) => d3.ascending(a, b));

viewof filtroDeptoVictima = Inputs.select(["Todos", ...departamentos], {label: "Departamento", value: "Todos"});

victimas = rows.filter((d) => filtroDeptoVictima === "Todos" || d.hecho_departamento === filtroDeptoVictima);

edadGrupo = (edad) => {
  if (!Number.isInteger(edad)) return "Sin dato";
  if (edad < 18) return "0-17";
  if (edad < 30) return "18-29";
  if (edad < 45) return "30-44";
  if (edad < 60) return "45-59";
  return "60+";
};

sexoVictima = (s) => s === "F" || s === "M" ? s : "Sin dato";
Total de victimas 
html`<div style="font-size:2.1rem;font-weight:700">${victimas.length.toLocaleString("es-BO")}</div>`

promEdad = d3.mean(victimas, (d) => d.victima_edad);
html`<div style="font-size:2.1rem;font-weight:700">${Number.isFinite(promEdad) ? promEdad.toFixed(1) : "N/D"}</div>`
Con discapacidad reportada 
conDiscap = victimas.filter((d) => d.victima_discapacidad).length;
porcDiscap = victimas.length > 0 ? (conDiscap / victimas.length) * 100 : NaN;
html`<div style="font-size:2.1rem;font-weight:700">${Number.isFinite(porcDiscap) ? `${porcDiscap.toFixed(1)}%` : "N/D"}</div>`
Sexo femenino 
femenino = victimas.filter((d) => sexoVictima(d.victima_sexo) === "F").length;
porcFem = victimas.length > 0 ? (femenino / victimas.length) * 100 : NaN;
html`<div style="font-size:2.1rem;font-weight:700">${Number.isFinite(porcFem) ? `${porcFem.toFixed(1)}%` : "N/D"}</div>`
Distribucion por sexo de victima 
sx = d3.rollups(victimas, (v) => v.length, (d) => sexoVictima(d.victima_sexo))
  .map(([sexo, casos]) => ({sexo, casos}))
  .sort((a, b) => d3.descending(a.casos, b.casos));

tablaDiscap = d3.rollups(victimas.filter((d) => d.victima_discapacidad), (v) => v.length, (d) => d.victima_discapacidad)

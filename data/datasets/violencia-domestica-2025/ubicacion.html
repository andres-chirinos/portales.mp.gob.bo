Ubicacion: departamentos, municipios y mapa
d3 = require("d3@7");
Plot = require("@observablehq/plot");

raw = await FileAttachment("VIOLENCIA_DOMESTICA 2025_publico__1.csv").csv({typed: false});

toText = (v) => (v ?? "").toString().trim();

toInt = (v) => {
  const n = Number.parseInt(toText(v), 10);
  return Number.isFinite(n) ? n : null;
};

rows = raw.map((d) => ({
  ...d,
  hecho_departamento: toText(d.hecho_departamento),
  hecho_municipio: toText(d.hecho_municipio),
  hecho_mes: toInt(d.hecho_mes)
}));

viewof filtroMesMapa = Inputs.select(
  ["Todos", ...d3.range(1, 13)],
  {
    label: "Mes",
    value: "Todos",
    format: (m) => m === "Todos" ? m : d3.timeFormat("%B")(new Date(2025, m - 1, 1))
  }
);

dataGeo = rows.filter((d) => filtroMesMapa === "Todos" || d.hecho_mes === filtroMesMapa);

depCounts = d3.rollups(dataGeo.filter((d) => d.hecho_departamento), (v) => v.length, (d) => d.hecho_departamento)
  .map(([departamento, casos]) => ({departamento, casos}))
  .sort((a, b) => d3.descending(a.casos, b.casos));

munCounts = d3.rollups(dataGeo.filter((d) => d.hecho_municipio), (v) => v.length, (d) => d.hecho_municipio)
  .map(([municipio, casos]) => ({municipio, casos}))
  .sort((a, b) => d3.descending(a.casos, b.casos));

coordsDepto = [
  {departamento: "PANDO", lat: -11.026, lon: -68.769},
  {departamento: "BENI", lat: -14.834, lon: -64.900},
  {departamento: "LA PAZ", lat: -16.495, lon: -68.133},

  {departamento: "COCHABAMBA", lat: -17.389, lon: -66.156},
  {departamento: "SANTA CRUZ", lat: -17.784, lon: -63.182},
  {departamento: "CHUQUISACA", lat: -19.047, lon: -65.259},
  {departamento: "POTOSI", lat: -19.583, lon: -65.753},
  {departamento: "TARIJA", lat: -21.533, lon: -64.733}
];

mapData = coordsDepto.map((d) => {
  const found = depCounts.find((x) => x.departamento === d.departamento);
  return {...d, casos: found ? found.casos : 0};
});

Mapa de puntos por departamento (aproximado) 
Plot.plot({
  width: 1000,
  height: 520,
  marginLeft: 64,
  marginRight: 32,
  marginTop: 40,
  marginBottom: 52,
  x: {label: "Longitud", domain: [-70.5, -57.5], grid: true},
  y: {label: "Latitud", domain: [-23.5, -9.5], grid: true},
  color: {scheme: "blues", legend: true, label: "Casos"},
  marks: [
    Plot.frame(),
    Plot.dot(mapData, {
      x: "lon",
      y: "lat",
      r: (d) => 4 + Math.sqrt(d.casos),
      fill: "casos",
      stroke: "#1b1f24",
      opacity: 0.9,
      tip: true,
      title: (d) => `${d.departamento}: ${d.casos}`
    }),
    Plot.text(mapData, {
      x: "lon",
      y: "lat",
      text: "departamento",
      dy: -10,
      fontSize: 11,
      fill: "#212529"
    })
  ]
})
